{{- /* Inline accent tokens for an element belonging to a section other than the page's own -- the home page, where every section appears at once and the body class cannot speak for all of them. Emits both inks rather than one. An inline --accent-ink would outrank any rule the dark palette could write, so the choice is left to CSS: the theme picks --accent-ink-light or --accent-ink-dark and assigns --accent-ink. Usage: {{ partial "func/accent-style.html" (dict "section" $s) }} {{ partial "func/accent-style.html" (dict "section" $s "full" true) }} `full` adds --accent-to and --on-accent, for elements that paint the accent as a surface rather than only as ink. */ -}} {{- $out := "" -}} {{- $s := .section -}} {{- with $s -}} {{- with .Params.accentFrom -}} {{- $ink := $s.Params.accentInk | default . -}} {{- $inkDark := $s.Params.accentInkDark | default (printf "color-mix(in srgb, %s 55%%, #fff)" .) -}} {{- if $.full -}} {{- $out = printf "--accent-from:%s;--accent-to:%s;--accent-ink-light:%s;--accent-ink-dark:%s;--on-accent:%s;" . ($s.Params.accentTo | default .) $ink $inkDark ($s.Params.onAccent | default "#fff") -}} {{- else -}} {{- $out = printf "--accent-from:%s;--accent-ink-light:%s;--accent-ink-dark:%s;" . $ink $inkDark -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
