{{- /* Title, description, canonical, social cards, icons, feed discovery and structured data. Everything is derived from the page or from site params, so the theme carries no site-specific values. */ -}} {{- $site := .Site -}} {{- $isHome := .IsHome -}} {{- /* Description: prefer what the author wrote, fall back to the generated summary, then to the site description. Trimmed to a length search engines and social cards will actually show. */ -}} {{- $desc := "" -}} {{- $authored := true -}} {{- with .Description }}{{ $desc = . }}{{ end -}} {{- if not $desc }}{{ with .Params.summary }}{{ $desc = . }}{{ end }}{{ end -}} {{- if not $desc }}{{ with .Summary }}{{ $desc = . }}{{ end }}{{ end -}} {{- /* Listing pages rarely have prose of their own. A synthesised line naming what is on the page beats repeating the site tagline on all of them. */ -}} {{- if not $desc -}} {{- $authored = false -}} {{- $count := len .Pages -}} {{- if and (eq .Kind "term") $count -}} {{- $desc = printf "%d %s on “%s” in %s." $count (cond (eq $count 1) "item" "items") .Title $site.Title -}} {{- else if and (eq .Kind "section") $count -}} {{- $desc = printf "%d %s in %s on %s." $count (cond (eq $count 1) "item" "items") .Title $site.Title -}} {{- else -}} {{- with $site.Params.description }}{{ $desc = . }}{{ end -}} {{- end -}} {{- end -}} {{- $desc = $desc | plainify | htmlUnescape | chomp -}} {{- /* Many summaries are written to sit beside the title on the page and read as a continuation of it ("Couchbase setup on AWS" + "EC2 instances"). That works in a card and fails in a search result, where the title is not adjacent. Lead with the title when the summary is short enough to be one of those -- but only for text the author actually wrote for this page. */ -}} {{- if and $authored (not $isHome) .Title (lt (len $desc) 60) -}} {{- if not (hasPrefix (lower $desc) (lower .Title)) -}} {{- $desc = printf "%s — %s" .Title $desc -}} {{- end -}} {{- end -}} {{- $desc = $desc | truncate 160 -}} {{- $title := cond $isHome $site.Title (printf "%s | %s" .Title $site.Title) -}} {{- /* Social image: page image, then a site-wide default, then the profile. $wide tracks whether the image is known to suit a wide card; a square image in a summary_large_image card is cropped to 1.91:1, which on a portrait means cutting the subject's head off. Default to the square `summary` card and only claim the large one when the shape is known. */ -}} {{- $img := "" -}} {{- $wide := false -}} {{- $res := false -}} {{- with .Params.image }} {{- $img = . -}} {{- $res = or ($.Resources.GetMatch .) (resources.Get .) -}} {{- end -}} {{- if not $img }}{{ with .Params.youtube_id }} {{- /* YouTube thumbnails are 16:9. */ -}} {{- $img = printf "https://img.youtube.com/vi/%s/maxresdefault.jpg" . -}} {{- $wide = true -}} {{- end }}{{ end -}} {{- if not $img }}{{ with $site.Params.ogImage }} {{- $img = . -}} {{- $res = resources.Get . -}} {{- end }}{{ end -}} {{- if not $img }}{{ with $site.Params.profileImage }} {{- $img = . -}} {{- $res = resources.Get . -}} {{- end }}{{ end -}} {{- /* Measure it when Hugo can. Files under static/ are not resources, so this simply stays unknown and the safe card type is used. */ -}} {{- $w := 0 }}{{ $h := 0 -}} {{- with $res }} {{- if and .Width .Height }} {{- $w = .Width }}{{ $h = .Height -}} {{- if ge (div (float .Width) (float .Height)) 1.5 }}{{ $wide = true }}{{ end -}} {{- end -}} {{- end -}} {{- with $site.Params.ogImageWidth }}{{ $w = . }}{{ end -}} {{- with $site.Params.ogImageHeight }}{{ $h = . }}{{ end -}} {{- if and $w $h (ne $h 0) }}{{ if ge (div (float $w) (float $h)) 1.5 }}{{ $wide = true }}{{ end }}{{ end -}} {{- with $img }}{{ if not (urls.Parse .).Scheme }}{{ $img = absURL . }}{{ end }}{{ end -}} {{ $title }} {{- with $desc }} {{- end }} {{- /* Open Graph */}} {{- with $desc }} {{- end }} {{- if and (eq .Kind "page") (not .Date.IsZero) }} {{- if and .Lastmod (ne .Lastmod.Year 1) }} {{- end }} {{- with partial "func/owning-section.html" . }}{{ with .Params.taxonomy }}{{ with (index $.Params .) }} {{- range . }} {{- end }} {{- end }}{{ end }}{{ end }} {{- else }} {{- end }} {{- with $img }} {{- if and $w $h (ne $w 0) (ne $h 0) }} {{- end }} {{- end }} {{- /* Twitter */}} {{- with $site.Params.twitterHandle }} {{- end }} {{- with $desc }} {{- end }} {{- with $img }} {{- end }} {{- /* Icons, declared by the site so the theme ships none of its own. */}} {{- range $site.Params.icons }} {{- end }} {{- with $site.Params.appleTouchIcon }} {{- end }} {{- /* Feed discovery: the site generates one but nothing pointed at it. */}} {{- $feed := .OutputFormats.Get "rss" }} {{- if not $feed }}{{ $feed = $site.Home.OutputFormats.Get "rss" }}{{ end }} {{- with $feed }} {{- end }} {{- /* Structured data */}} {{- if $isHome }} {{- $sameAs := slice }} {{- range $site.Params.social }}{{ $sameAs = $sameAs | append .url }}{{ end }} {{- else if and (eq .Kind "page") (not .Date.IsZero) }} {{- end }}
